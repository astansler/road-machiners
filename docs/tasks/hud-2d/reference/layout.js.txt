// ROAM HUD v4 \u00b7 ten cab layouts built from the kit.
import { STUB as THREE } from './three-stub.js';
import { Panel, Hud, box, cyl, ball, cable, coil, shade, F_MONO, F_SLAB } from './kit-flat.js';
import * as D from './screens.js';

/* state helpers */
const swSt = k => s => s.locked?.[k] ? 'locked' : s.hot?.[k] ? 'hot' : (k === 'cool' || k === 'aid') ? 'on' : s.sw[k] ? 'on' : 'off';
const fireSt = k => s => s.fire[k] ? 'on' : 'off';
const SW = [['route', 'R', 'ROUTE'], ['patch', 'P', 'PATCH'], ['lights', 'L', 'LIGHTS'], ['overdrive', 'O', 'OVERDR']];
const BTN = [['cool', 'G', 'COOL'], ['aid', 'U', 'AID']];
const FIRE = [['auto', 'AUTO'], ['all', 'ALL'], ['show', 'ARCS']];
const NAV = [['C', 'DRIVER'], ['I', 'TRUCK'], ['M', 'MAP'], ['J', 'JOURNAL'], ['ESC', 'MENU']];
const fuelCol = (u, s) => s.fuel <= 10 ? '#ff5533' : s.fuel <= 25 ? '#ffd75e' : (u < 0.17 ? '#ff5533' : u < 0.42 ? '#ffd75e' : '#6dff8f');
const engCol = u => u > 0.93 ? '#ff5533' : u > 0.79 ? '#ffd75e' : '#ffab2e';
const V = (x, y, z) => new THREE.Vector3(x, y, z);

/* what a press does. The mock state is edited in place so the cab answers. */
const say = (s, msg, tone = '') => s.log.unshift([s.time, msg, tone]);
export const ACT = {
  sw: k => s => { if (s.locked?.[k]) return; s.sw[k] = s.sw[k] ? 0 : 1; if (s.hot) delete s.hot[k]; },
  cool: s => { if (s.locked?.cool || s.supplies < 1) return; s.eng = Math.max(40, s.eng - 30); s.supplies--; say(s, 'Engine doused. One supply gone.', 'good'); if (s.eng < 95) (s.locked ||= {}).cool = 1; },
  aid: s => { if (s.locked?.aid || s.supplies < 1) return; s.health = Math.min(100, s.health + 25); s.supplies--; say(s, 'Driver patched up. One supply gone.', 'good'); if (s.health >= 100) (s.locked ||= {}).aid = 1; },
  fire: k => s => { s.fire[k] = s.fire[k] ? 0 : 1; },
  turn: s => { s.turn = ((s.turn ?? 2) + 1) % 3; },
  end: s => { let [h, m] = s.time.split(':').map(Number); m += 6; if (m >= 60) { m -= 60; h = (h + 1) % 24; } s.time = h + ':' + String(m).padStart(2, '0'); s.fuel = Math.max(0, Math.round((s.fuel + s.fuelRate) * 10) / 10); s.heading = (s.heading + 7) % 360; },
};
const OPEN = { navJ: 'log', navI: 'truck' };
const press = id => s => { // screen keys open the menus that exist; the rest only click
  s.pressed = id; s.pressedAt = performance.now();
  if (OPEN[id]) s.open = s.open === OPEN[id] ? null : OPEN[id];
  else if (id === 'ctx' && s.objective.ctx && /^Enter/.test(s.objective.ctx[1])) s.open = s.open === 'town' ? null : 'town';
  else if (id === 'navESC') s.open = null;
};
const pressed = id => s => s.pressed === id && performance.now() - s.pressedAt < 260 ? 'on' : 'off';
const swAct = k => k === 'cool' ? ACT.cool : k === 'aid' ? ACT.aid : ACT.sw(k);
export const HOTKEYS = { r: ACT.sw('route'), p: ACT.sw('patch'), l: ACT.sw('lights'), o: ACT.sw('overdrive'), g: ACT.cool, u: ACT.aid, ' ': ACT.end, c: press('navC'), i: press('navI'), m: press('navM'), j: press('navJ'), escape: press('navESC'), e: press('ctx') };

/* control clusters, shared by the layouts */
function toggles(p, x, y, dx, o = {}) { SW.forEach(([k, h, l], i) => p.toggle(x + i * dx, y, { label: l, hot: h, get: swSt(k), click: swAct(k), guard: k === 'overdrive' ? (o.guard ?? true) : false, color: k === 'overdrive' ? '#ffab2e' : o.lamp, textColor: o.text })); }
function swKeys(p, x, y, w, h, dx, o = {}) { SW.forEach(([k, hk, l], i) => p.key(x + i * dx, y, w, h, { label: l, hot: hk, get: swSt(k), click: swAct(k), color: k === 'overdrive' ? '#ff8a3a' : (o.color || '#ffab2e'), led: o.led, cap: o.cap })); }
function shots(p, x, y, w, h, dx, o = {}) { BTN.forEach(([k, hk, l], i) => p.key(x + (o.vert ? 0 : i * dx), y - (o.vert ? i * dx : 0), w, h, { label: l, hot: hk, get: swSt(k), click: swAct(k), color: k === 'cool' ? '#7fd6e8' : '#ff7a66', cap: o.cap })); }
function fireKeys(p, x, y, w, h, dx, o = {}) { FIRE.forEach(([k, l], i) => p.key(x + (o.vert ? 0 : i * dx), y - (o.vert ? i * dx : 0), w, h, { label: l, get: fireSt(k), click: ACT.fire(k), color: o.color || '#ff5a3a', cap: o.cap, led: o.led })); }
function navKeys(p, x, y, w, h, dx, o = {}) { NAV.forEach(([hk, l], i) => p.key(x + (o.vert ? 0 : i * dx), y - (o.vert ? i * dx : 0), w, h, { label: o.short ? hk : l, hot: o.short ? null : hk, get: pressed('nav' + hk), click: press('nav' + hk), color: '#e9dfc4', dimText: o.dim || '#cfc7b0', cap: o.cap, dark: o.dark })); }
function endTurn(p, x, y, w, h, o = {}) { return p.key(x, y, w, h, { label: 'END TURN', hot: 'SPACE', get: () => 'on', click: ACT.end, color: o.color || '#e9dfc4', cap: o.cap, pop: 0.13 }); }
function ctxKey(p, x, y, w, h, o = {}) { return p.key(x, y, w, h, { label: s => s.objective.ctx ? s.objective.ctx[1].toUpperCase() : '—', hot: null, get: s => s.objective.ctx ? 'on' : 'off', click: press('ctx'), color: o.color || '#ffab2e', cap: o.cap }); }
function ctxHot(p, x, y, o = {}) { return p.screen(x, y, 0.3, 0.3, { pal: o.pal || 'amber', pop: 0.05, frame: 0.04, every: 2, draw: (c, W, H, s, t, P) => { c.font = `700 ${H * 0.6}px ${F_MONO}`; c.textAlign = 'center'; c.textBaseline = 'middle'; c.fillStyle = P.fg; c.fillText(s.objective.ctx ? s.objective.ctx[0] : '', W / 2, H * 0.54); } }); }
function turnKnob(p, x, y, r = 0.17) { return p.knob(x, y, r, { marks: ['1×', '2×', '4×'], get: s => s.turn ?? 2, click: ACT.turn, label: 'TURN' }); }

/* shared screen shorthands */
const PALE = '#e9dfc4';
const speedGauge = (p, x, y, r, o = {}) => p.gauge(x, y, r, { min: 0, max: 90, step: 30, sub: 3, label: 'KM/H', get: s => s.speed, read: s => String(s.speed), zones: s => [[s.limit, 90, s.limit < 54 ? '#ff5533' : '#c9b9a0']], ...o });
const fuelGauge = (p, x, y, r, o = {}) => p.gauge(x, y, r, { min: 0, max: 60, step: 30, sub: 3, label: 'FUEL L', get: s => s.fuel, read: s => String(s.fuel), a0: 200, a1: -20, zones: () => [[0, 10, '#ff5533'], [10, 25, '#e3b23a']], ...o });
const engGauge = (p, x, y, r, o = {}) => p.gauge(x, y, r, { min: 40, max: 120, step: 40, sub: 4, label: 'ENG °C', get: s => s.eng, read: s => String(s.eng), a0: 200, a1: -20, zones: () => [[95, 112, '#e3b23a'], [112, 120, '#ff5533']], ...o });
const msWin = (p, x, y, w = 0.9, h = 0.26, pal = 'amber') => p.screen(x, y, w, h, { pal, pop: 0.04, frame: 0.045, draw: D.ms, every: 2 });
const clockWin = (p, x, y, w = 0.9, h = 0.26, pal = 'red') => p.screen(x, y, w, h, { pal, pop: 0.04, frame: 0.045, draw: D.clock });
const logWin = (p, x, y, w, h = 0.22, pal = 'lcd') => p.screen(x, y, w, h, { pal, pop: 0.04, frame: 0.045, every: 2, draw: D.logLine });
const pagerWin = (p, x, y, w, h = 0.5, pal = 'lcd') => p.screen(x, y, w, h, { pal, pop: 0.05, frame: 0.05, every: 2, draw: D.pager });
const tapeWin = (p, x, y, w, h = 0.26, pal = 'amber') => p.screen(x, y, w, h, { pal, pop: 0.04, frame: 0.045, draw: (c, W, H, s, t, P) => D.headingTape(c, W * 0.03, H * 0.08, W * 0.94, H * 0.84, s, P) });
