const TONE = { bad: '#b3321f', good: '#2f6b3a', warn: '#9a6a12', dim: '#8d8371', '': '#2a2622' };
const FILTERS = [['all', 'ALL', () => true], ['bad', 'TROUBLE', l => l[2] === 'bad' || l[2] === 'warn'], ['good', 'GOOD NEWS', l => l[2] === 'good'], ['dim', 'CHATTER', l => l[2] === 'dim']];
function history(data, s) { // today's lines first, then everything the other mock situations logged
  const seen = new Set(), out = [];
  for (const l of [...s.log, ...Object.values(data.situations).flatMap(x => x.log)]) { const k = l[0] + l[1]; if (!seen.has(k)) { seen.add(k); out.push(l); } }
  return out.sort((a, b) => { const m = t => { const [h, mm] = t.split(':').map(Number); return h * 60 + mm; }; return m(b[0]) - m(a[0]); });
}
function logMenu({ data }) {
  const hud = new Hud(), K = '#d8ceb2';
  const p = hud.add(new Panel(9.0, 5.3, { color: K, seed: 201, d: 0.6, r: 0.14, wear: 0.7 }).at(0, 1.55, 1.6, -0.06));
  const rows = s => { const f = FILTERS.find(x => x[0] === ui(s).filter)[2], out = []; for (const l of history(data, s).filter(f)) wrap(l[1], 50).forEach((txt, i) => out.push([i ? '' : l[0], txt, l[2]])); return out; };
  const VIS = 14;
  p.screen(-1.5, -0.2, 5.5, 4.4, { pal: 'paper', pop: 0.05, frame: 0.1, bezel: '#3a3b40', glass: false, round: 0.02, every: 1, res: 1.2, draw: (c, W, H, s, t, P) => {

    for (let y = H * 0.03; y < H; y += H * 0.055) { c.beginPath(); c.arc(W * 0.0225, y, W * 0.008, 0, 7); c.arc(W * 0.9775, y, W * 0.008, 0, 7); c.fill(); }
    T(c, 'ROAD LOG \u00b7 DAY ' + s.day, W * 0.07, H * 0.055, H * 0.045, P.fg, 'left', 700); T(c, FILTERS.find(x => x[0] === ui(s).filter)[1], W * 0.93, H * 0.055, H * 0.034, P.dim, 'right');

    T(c, (sc + 1) + '–' + Math.min(all.length, sc + VIS) + ' of ' + all.length, W * 0.93, H * 0.965, H * 0.03, P.dim, 'right');

  } });
  roller(p, -1.5, 2.22, 5.6);
  p.dymo(2.95, 2.25, 'SHOW');
  FILTERS.forEach(([k, label], i) => pick(p, 2.95, 1.8 - i * 0.5, 2.3, 0.42, label, s => ui(s).filter === k, s => { ui(s).filter = k; ui(s).scroll = 0; }));
  p.dymo(2.95, -0.28, 'ROLL');
  p.key(2.37, -0.72, 1.1, 0.5, { label: 'UP', get: () => 'off', click: s => { ui(s).scroll = Math.max(0, ui(s).scroll - 5); }, color: '#e9dfc4', dimText: '#e9dfc4' });
  p.key(3.53, -0.72, 1.1, 0.5, { label: 'DOWN', get: () => 'off', click: s => { ui(s).scroll += 5; }, color: '#e9dfc4', dimText: '#e9dfc4' });
  p.screen(2.95, -1.5, 2.3, 0.78, { pal: 'lcd', pop: 0.05, draw: D.contracts, every: 2 });
  closeKey(p, 2.95, -2.25, 2.3, 0.44);
  p.plate(-3.4, -2.52, 1.3, 0.16, ['GUIDESTAR TELEPRINTER', 'RELAY 31']); p.tape(0.3, -2.52, 1.4, 0.14, 'ribbon is nearly dry', 0.01); p.decal(-1.6, -2.48, 0.13, 'guild', 0.2); p.broken(-1, 1, 0.3); p.bolts(0.14);
