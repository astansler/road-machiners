/* seven-segment digits */
const SEG = { 0: 'abcdef', 1: 'bc', 2: 'abged', 3: 'abgcd', 4: 'fgbc', 5: 'afgcd', 6: 'afgedc', 7: 'abc', 8: 'abcdefg', 9: 'abfgcd', '-': 'g', L: 'fed', C: 'afed', d: 'bcdeg', E: 'afged', H: 'fbgec', ' ': '' };
export function seg7(c, str, x, y, h, on, off) { // right edge at x when align right is wanted: use seg7w first
  const w = h * 0.52, th = h * 0.12, gap = h * 0.16; let cx = x;
  const bar = (x1, y1, x2, y2) => { c.beginPath(); c.moveTo(x1, y1); c.lineTo(x2, y2); c.stroke(); };
  c.lineCap = 'round'; c.lineWidth = th;
  for (const ch of String(str)) {
    if (ch === ':' || ch === '.') { c.fillStyle = on; if (ch === ':') { c.fillRect(cx, y + h * 0.28, th, th); c.fillRect(cx, y + h * 0.66, th, th); } else c.fillRect(cx, y + h - th, th, th); cx += th + gap; continue; }
    const L = cx + th / 2, R = cx + w - th / 2, Tp = y + th / 2, M = y + h / 2, B = y + h - th / 2, i = th * 0.75;
    const segs = { a: [L + i, Tp, R - i, Tp], b: [R, Tp + i, R, M - i], c: [R, M + i, R, B - i], d: [L + i, B, R - i, B], e: [L, M + i, L, B - i], f: [L, Tp + i, L, M - i], g: [L + i, M, R - i, M] };
    const lit = SEG[ch] ?? '';
    for (const k in segs) { c.strokeStyle = lit.includes(k) ? on : off; if (lit.includes(k) || off) bar(...segs[k]); }
    cx += w + gap;
  }
  return cx - gap - x;
}
export const seg7w = (str, h) => [...String(str)].reduce((a, ch) => a + ((ch === ':' || ch === '.') ? h * 0.12 : h * 0.52) + h * 0.16, 0) - h * 0.16;
export function segR(c, str, xr, y, h, on, off) { return seg7(c, str, xr - seg7w(str, h), y, h, on, off); }

export function headingTape(c, x, y, w, h, s, P) {
  c.save(); c.beginPath(); c.rect(x, y, w, h); c.clip();
  const ppd = w / 110, cx = x + w / 2;
  for (let d = Math.floor(s.heading / 10) * 10 - 60; d <= s.heading + 60; d += 10) {
    const px = cx + (d - s.heading) * ppd, dd = ((d % 360) + 360) % 360, big = dd % 30 === 0;
    c.fillStyle = big ? P.fg : P.dim; c.fillRect(px - 1, y + h * (big ? 0.5 : 0.68), 2, h * (big ? 0.5 : 0.32));
    if (big) T(c, { 0: 'N', 90: 'E', 180: 'S', 270: 'W' }[dd] ?? String(dd / 10).padStart(2, '0'), px, y + h * 0.26, h * 0.42, dd % 90 ? P.dim : P.fg, 'center');
  }
  c.restore(); c.fillStyle = P.fg; c.beginPath(); c.moveTo(cx, y + h * 0.55); c.lineTo(cx - h * 0.2, y + h); c.lineTo(cx + h * 0.2, y + h); c.fill();
}
export function speedBar(c, x, y, w, h, s, P) {
  const k = v => x + w * v / s.top; c.strokeStyle = P.dim; c.lineWidth = 2; c.strokeRect(x, y, w, h);
  c.fillStyle = P.fg; c.fillRect(x + 3, y + 3, Math.max(0, k(s.speed) - x - 6), h - 6);
  c.save(); c.beginPath(); c.rect(k(s.limit), y, x + w - k(s.limit), h); c.clip(); c.strokeStyle = s.limit < s.top * 0.6 ? P.hot : P.dim; c.lineWidth = 3;
  for (let i = -h; i < w; i += 9) { c.beginPath(); c.moveTo(x + i, y + h); c.lineTo(x + i + h, y); c.stroke(); } c.restore();
  for (let v = 0; v <= s.top; v += 10) { c.fillStyle = P.dim; c.fillRect(k(v) - 1, y + h + 2, 2, v % 30 ? 4 : 8); }
  c.fillStyle = P.warn; c.beginPath(); c.moveTo(k(s.hold), y - 2); c.lineTo(k(s.hold) - 6, y - 11); c.lineTo(k(s.hold) + 6, y - 11); c.fill();
}

const PART = { 'Stock engine': 'engine', 'MG turret': 'turret', Cab: 'cab', 'Rebar cage': 'cage', Transmission: 'trans' };
export function truck(c, x, y, w, h, s, t, P, horiz) { // top-down grid of what is fitted; nose up, or nose right when horiz
  const hp = Object.fromEntries(s.parts.map(p => [p.k, p.hp]));
  const cols = GRID.cols, rows = GRID.rows, cell = horiz ? Math.min(w / (rows + 1), h / (cols + 2)) : Math.min(w / (cols + 2), h / (rows + 1));
  const ox = horiz ? x + (w - rows * cell) / 2 : x + (w - cols * cell) / 2, oy = horiz ? y + (h - cols * cell) / 2 : y + (h - rows * cell) / 2;
  const rect = (col, row, cw, ch, key, plain) => {
    const X = horiz ? ox + (rows - row - ch + 1) * cell : ox + (col - 1) * cell, Y = horiz ? oy + (col - 1) * cell : oy + (row - 1) * cell, Wd = (horiz ? ch : cw) * cell, Hd = (horiz ? cw : ch) * cell;
    const v = key ? hp[key] ?? 100 : 100, l = v < 40 ? 2 : v < 70 ? 1 : 0;
    c.lineWidth = 2; c.strokeStyle = plain ? P.dim : lc(P, l); c.strokeRect(X + 2, Y + 2, Wd - 4, Hd - 4);
    if (l === 2 ? blink(t) : l === 1) { c.fillStyle = lc(P, l); c.globalAlpha = l === 2 ? 0.85 : 0.4; c.fillRect(X + 2, Y + 2, Wd - 4, Hd - 4); c.globalAlpha = 1; }
    else if (!plain) { c.fillStyle = P.fg; c.globalAlpha = 0.13; c.fillRect(X + 2, Y + 2, Wd - 4, Hd - 4); c.globalAlpha = 1; }
  };
  c.strokeStyle = P.dim; c.lineWidth = 1; c.setLineDash([3, 4]);
  if (horiz) c.strokeRect(ox, oy, rows * cell, cols * cell); else c.strokeRect(ox, oy, cols * cell, rows * cell); c.setLineDash([]);
  for (const it of GRID.items) rect(it.col, it.row, it.w, it.h, PART[it.name], !PART[it.name]);
  rect(2, 8, 2, 1, 'tank');
  for (const [col, row] of [[0, 2], [5, 2], [0, 10], [5, 10]]) rect(col, row, 1, 2, 'wheels');
}

export function radar(c, W, H, s, t, P) {
  const R = Math.min(W, H) * 0.46, cx = W / 2, cy = H / 2;
  c.strokeStyle = P.dim; c.lineWidth = 1.5; for (const k of [1 / 3, 2 / 3, 1]) { c.beginPath(); c.arc(cx, cy, R * k, 0, 7); c.stroke(); }
  c.beginPath(); c.moveTo(cx - R, cy); c.lineTo(cx + R, cy); c.moveTo(cx, cy - R); c.lineTo(cx, cy + R); c.stroke();
  const sw = t * 1.1; c.fillStyle = P.fg; c.globalAlpha = 0.16; c.beginPath(); c.moveTo(cx, cy); c.arc(cx, cy, R, sw - 0.5, sw); c.fill(); c.globalAlpha = 1;
  c.strokeStyle = P.fg; c.lineWidth = 2; c.beginPath(); c.moveTo(cx, cy); c.lineTo(cx + Math.cos(sw) * R, cy + Math.sin(sw) * R); c.stroke();
  const ha = s.heading * Math.PI / 180; c.fillStyle = P.fg; c.beginPath(); c.moveTo(cx + Math.sin(ha) * R * 0.14, cy - Math.cos(ha) * R * 0.14); c.lineTo(cx + Math.sin(ha + 2.5) * R * 0.09, cy - Math.cos(ha + 2.5) * R * 0.09); c.lineTo(cx + Math.sin(ha - 2.5) * R * 0.09, cy - Math.cos(ha - 2.5) * R * 0.09); c.fill();
  for (const [name, , dist, stance, sel] of s.contacts) {
    const k = Math.min(1, Math.log10(Math.max(60, meters(dist)) / 60) / Math.log10(50)), a = (hash(name) % 628) / 100, x = cx + Math.cos(a) * R * k * 0.95, y = cy + Math.sin(a) * R * k * 0.95, z = R * 0.075;
    const col = stance === 'hostile' ? P.hot : stance === 'friendly' ? P.fg : P.warn; c.fillStyle = col; c.strokeStyle = col; c.lineWidth = 2;
    if (stance === 'hostile') { c.beginPath(); c.moveTo(x, y - z * 1.3); c.lineTo(x + z, y); c.lineTo(x, y + z * 1.3); c.lineTo(x - z, y); c.fill(); }
    else if (stance === 'friendly') { c.beginPath(); c.arc(x, y, z * 0.9, 0, 7); c.fill(); }
    else if (stance) c.strokeRect(x - z * 0.8, y - z * 0.8, z * 1.6, z * 1.6); else { c.beginPath(); c.moveTo(x - z, y - z); c.lineTo(x + z, y + z); c.moveTo(x + z, y - z); c.lineTo(x - z, y + z); c.stroke(); }
    if (sel && blink(t)) { c.strokeStyle = P.hot; c.strokeRect(x - z * 2, y - z * 2, z * 4, z * 4); }
  }
}
