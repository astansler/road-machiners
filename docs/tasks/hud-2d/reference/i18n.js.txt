];
export const LANGS = { en: null, ru: { words: RU, rx: RU_RX } };
export function makeTr(lang) {
  const L = LANGS[lang]; if (!L) return null;
  return (s, where) => { if (where && (where + ':' + s) in L.words) return L.words[where + ':' + s]; if (s in L.words) return L.words[s]; for (const [rx, to] of L.rx) if (rx.test(s)) return s.replace(rx, to); return s; };
}
