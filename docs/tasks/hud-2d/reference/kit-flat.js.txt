/* colour helpers, same results as the three.js versions (shade: sRGB HSL; mix: linear light) */
const hex2 = h => { h = h.replace('#', ''); if (h.length === 3) h = [...h].map(c => c + c).join(''); return [0, 2, 4].map(i => parseInt(h.slice(i, i + 2), 16) / 255); };
const toHex = a => '#' + a.map(v => Math.round(Math.min(1, Math.max(0, v)) * 255).toString(16).padStart(2, '0')).join('');
export function shade(hex, l, s = 0) {
  const [r, g, b] = hex2(hex), mx = Math.max(r, g, b), mn = Math.min(r, g, b); let H = 0, S = 0; const L = (mx + mn) / 2;
  if (mx !== mn) { const d = mx - mn; S = L <= 0.5 ? d / (mx + mn) : d / (2 - mx - mn); H = mx === r ? (g - b) / d + (g < b ? 6 : 0) : mx === g ? (b - r) / d + 2 : (r - g) / d + 4; H /= 6; }
  const h = ((H % 1) + 1) % 1, S2 = Math.min(1, Math.max(0, S + s)), L2 = Math.min(1, Math.max(0, L + l));
  if (S2 === 0) return toHex([L2, L2, L2]);
  const q = L2 <= 0.5 ? L2 * (1 + S2) : L2 + S2 - L2 * S2, p = 2 * L2 - q;
  const hue = t => { if (t < 0) t += 1; if (t > 1) t -= 1; return t < 1 / 6 ? p + (q - p) * 6 * t : t < 1 / 2 ? q : t < 2 / 3 ? p + (q - p) * 6 * (2 / 3 - t) : p; };
  return toHex([hue(h + 1 / 3), hue(h), hue(h - 1 / 3)]);
}
const lin = c => c < 0.04045 ? c * 0.0773993808 : Math.pow(c * 0.9478672986 + 0.0521327014, 2.4);
const srgb = c => c < 0.0031308 ? c * 12.92 : 1.055 * Math.pow(c, 0.41666) - 0.055;
export function mix(a, b, k) { const A = hex2(a).map(lin), B = hex2(b).map(lin); return toHex(A.map((v, i) => srgb(v + (B[i] - v) * k))); }
export function rng(seed) {
  let a = seed >>> 0;
  return () => { a |= 0; a = a + 0x6D2B79F5 | 0; let t = Math.imul(a ^ a >>> 15, 1 | a); t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t; return ((t ^ t >>> 14) >>> 0) / 4294967296; };
}

let REC = null; // the live parts of the Hud being built, in call order
const rec = (kind, panel, a, o) => { const e = { kind, a, o, panel }; if (REC) REC.push(e); return STUB; };

export class Panel {
  constructor(w, h, o = {}) {
    this.w = w; this.h = h; this.o = Object.assign({ d: 0.3, r: 0.08, color: '#86a88c', seed: 1, wear: 0.6, text: INK, bezel: '#2b2c30', cap: '#3b3c40', metal: '#b9b3a3' }, o);
    this.z = this.o.d / 2; this.g = STUB; this.kids = [];
  }
  X(x) { return (x + this.w / 2) * PPU; } Y(y) { return (this.h / 2 - y) * PPU; }
  at() { return this; } add(obj) { return obj; } paint() { return this; } finish() { return this; }
  text() { return this; } recess() { return this; } seam() { return this; } patch() { return this; } rub() { return this; }
  hazard() { return this; } plate() { return this; } tape() { return this; } scrawl() { return this; } decal() { return this; }
  dymo() { return this; } stain() { return this; } broken() { return this; } stripe() { return this; }
  bolt() { return this; } bolts() { return this; } hood() { return STUB; }
  child(p) { this.kids.push(p); return p; }
  all() { return [this, ...this.kids.flatMap(k => k.all())]; }
  /* live parts */
  screen(x, y, w, h, o) { return rec('screen', this, [x, y, w, h], o); }
  paper(x, y, w, h, o) { return this.screen(x, y, w, h, { pal: 'paper', pop: 0.012, frame: 0, glass: false, round: 0.01, bezel: '#cfc5aa', every: 9, ...o }); }
  gauge(x, y, r, o) { return rec('gauge', this, [x, y, r], o); }
  key(x, y, w, h, o) { return rec('key', this, [x, y, w, h], o); }
  toggle(x, y, o) { return rec('toggle', this, [x, y], o); }
  knob(x, y, r, o = {}) { return rec('knob', this, [x, y, r], o); }
  ledbar(x, y, w, h, n, o) { return rec('ledbar', this, [x, y, w, h, n], o); }
  lamp(x, y, r, o) { return rec('lamp', this, [x, y, r], o); }
}

export class Hud {
  constructor() { this.root = STUB; this.panels = []; this.parts = REC = []; }
  add(p) { this.panels.push(p); return p; }
  extra(obj) { return obj; }
  seal() { REC = null; return this; }
  update() {} dispose() {}
}
