// ROAM HUD v4 \u00b7 2D runtime. Shows a baked cab with plain DOM: no WebGL, no three.js.
//
//   still picture   one <img> per layer (cab, menu), rendered once from the fixed camera
//   screens, dials, key tops
//                   <canvas> drawn by the same functions as in 3D (screens.js), put in place
//                   with a CSS matrix3d (the exact perspective of that face)
//   levers, knobs, pressed keys
//                   cut-out frames from the atlas, one shown at a time
//   lamps, LED cells
//                   colour masks from the atlas, coloured with background-color
//   needles         a flat needle sprite laid into the dial plane, turned with CSS rotate
//   labels          real DOM text (labels, label tape, handwriting) set on the panel plane with matrix3d,
//                   in the lit colour of that spot; pass { tr } to translate it
//
// Everything is in a 2560 × 1440 design space that is scaled to the host as one piece.

const el = (tag, cls, parent) => { const e = document.createElement(tag); if (cls) e.className = cls; if (parent) parent.append(e); return e; };
const place = (e, x, y, w, h) => { e.style.left = x + 'px'; e.style.top = y + 'px'; e.style.width = w + 'px'; e.style.height = h + 'px'; return e; };
const m3d = m => `matrix3d(${m.join(',')})`;

/* 3 × 3 homography helpers for hit tests on faces */
const H3 = m => [m[0], m[4], m[12], m[1], m[5], m[13], m[3], m[7], m[15]];
function inv3(a) {
  const [a0, a1, a2, a3, a4, a5, a6, a7, a8] = a, b01 = a8 * a4 - a5 * a7, b11 = -a8 * a3 + a5 * a6, b21 = a7 * a3 - a4 * a6, d = a0 * b01 + a1 * b11 + a2 * b21;
  return [b01, -a8 * a1 + a2 * a7, a5 * a1 - a2 * a4, b11, a8 * a0 - a2 * a6, -a5 * a0 + a2 * a3, b21, -a7 * a0 + a1 * a6, a4 * a0 - a1 * a3].map(v => v / d);
}
const apply3 = (h, x, y) => { const w = h[6] * x + h[7] * y + h[8]; return [(h[0] * x + h[1] * y + h[2]) / w, (h[3] * x + h[4] * y + h[5]) / w]; };
function inPoly(p, x, y) { let c = false; for (let i = 0, j = p.length - 1; i < p.length; j = i++) if ((p[i][1] > y) !== (p[j][1] > y) && x < (p[j][0] - p[i][0]) * (y - p[i][1]) / (p[j][1] - p[i][1]) + p[i][0]) c = !c; return c; }

      const xs = h.poly.map(p => p[0]), ys = h.poly.map(p => p[1]), x0 = Math.min(...xs), y0 = Math.min(...ys);
      const d = h.div = place(el('div', 'fl-hit', gE), x0, y0, Math.max(...xs) - x0, Math.max(...ys) - y0);
      d.style.clipPath = `polygon(${h.poly.map(([x, y]) => `${(x - x0).toFixed(1)}px ${(y - y0).toFixed(1)}px`).join(',')})`;
    }
  }
  // switch the language: live panel text now, keys and dials on the next update(…, force)
  translate(tr) {
    this.tr = tr;
    for (const S of this.spans) { // a longer translation of a plain label narrows to 1.25 × the original width; label tape grows instead
      const { sp, it, px, place2 } = S; sp.textContent = it.s; sp.style.fontSize = ''; if (!S.w0) S.w0 = sp.offsetWidth;
      const t = tr ? tr(it.s, it.dymo ? 'dymo' : 'label') : it.s; sp.textContent = t; let k = 1;
      if (t !== it.s && !it.dymo && S.w0) { const w1 = sp.offsetWidth; if (w1 > S.w0 * 1.25) k = Math.max(0.6, S.w0 * 1.25 / w1); }
      sp.style.fontSize = k < 1 ? px * k + 'px' : ''; place2(k);
    }
    for (const it of this.items) it.last = -9;
  }
  // a screen as a slot for the game's own interface: an element in that face's perspective, cw × ch CSS px
  slot(id) {
    const S = this.screens.find(x => x.id === id); if (!S) throw new Error('flat: no screen ' + id + ' (have ' + this.screens.map(x => x.id).join(', ') + ')');
    if (!S.slot) { const d = el('div', 'fl-slot', S.cv.parentNode); d.style.width = S.cv.style.width; d.style.height = S.cv.style.height; d.style.transform = S.cv.style.transform; d.style.borderRadius = S.cv.style.borderRadius; d.style.background = S.bg;
      S.cv.style.display = 'none'; S.it.fn = null; if (S.hit) { S.hit.off = true; if (S.hit.div) S.hit.div.remove(); } S.slot = d; }
    return S.slot;
  }
  update(s, dt, force) {
    this.t += dt;
    for (const it of this.items) { if (it.tick) it.tick(dt); if (it.fn && (force || this.t - it.last >= it.every)) { it.last = this.t; it.fn(s, this.t); } }
  }
  pick(x, y) { // design-space point → st => void, or null
    for (const h of this.hits) { if (h.off) continue; const r = h.test(x, y); if (r) return h.fn(r); }
    return null;
  }
}

/* the design space scaled into a host element (the game view) */
export class FlatStage {
  constructor(host) {
    if (!styled) { styled = true; el('style', '', document.head).textContent = CSS; }
    this.host = host; this.root = el('div', 'fl-stage', host); this.space = el('div', 'fl-space', this.root); this.k = 1; this.dx = 0; this.dy = 0;
    this.ro = new ResizeObserver(() => this.fit()); this.ro.observe(host); this.fit();
  }
  fit() { this.k = this.host.clientWidth / DESIGN_W; this.apply(); }
  shift(dx, dy) { this.dx = dx; this.dy = dy; this.apply(); } // design px
  apply() { this.space.style.transform = `scale(${this.k}) translate(${this.dx}px,${this.dy}px)`; }
  add(layer) { this.space.append(layer.root); return layer; }
  toDesign(clientX, clientY) { const r = this.host.getBoundingClientRect(); return [(clientX - r.left) / this.k - this.dx, (clientY - r.top) / this.k - this.dy]; }
  destroy() { this.ro.disconnect(); this.root.remove(); }
}
